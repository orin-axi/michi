     ```rust
     pub(crate) fn escape_value(v: &str) -> std::borrow::Cow<'_, str> {
         if v.contains(',') || v.contains('"') || v.contains('\n') || v.contains('\r') {
             let mut out = String::with_capacity(v.len() * 2 + 2);
             out.push('"');
             for ch in v.chars() {
                 if ch == '"' { out.push('\\'); }
                 out.push(ch);
             }
             out.push('"');
             std::borrow::Cow::Owned(out)
         } else {
             std::borrow::Cow::Borrowed(v)
         }
     }
     ```
