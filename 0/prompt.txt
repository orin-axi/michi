   pub enum AgentError {
       Http { status: u16, message: String, retryable: bool, retry_after: Option<Duration> },

       CyclicDependency { cycle: Vec<String> },
       Cancelled,
       Cache(String),
       Io(#[from] std::io::Error),
   }
   
   // OutputSink — surface-agnostic event bus
   pub enum AgentEvent {
       StepStarted { id, name, level },
       StepCompleted { id, duration },
       StepSkipped { id, reason },
       StepFailed { id, error, retryable },
       RetryScheduled { label, attempt, max, wait },
       CircuitOpened { name, retry_after },
       PipelineFinished(PipelineResult),
       Data(serde_json::Value),
       Warn(String), Info(String),
   }
   pub trait OutputSink: Send + Sync { fn emit(&self, event: AgentEvent); }
   
   // Step execution
   pub struct Step {
       pub id: String,
       pub name: String,
       pub level: u32,
       pub depends_on: Vec<String>,
       pub idempotency_key: Option<String>,
       pub best_effort: bool,
       pub run: Arc<dyn StepFn>,
   }
   
   // Checkpoint — pure data, no I/O in core
   pub struct Checkpoint { pipeline_id, run_id, created_at, completed, step_outputs, failed }
   pub trait CheckpointStore: Send + Sync { save/load/clear }
   
   // Pipeline result
   pub enum RunStatus { Complete, Partial, Failed }
   pub struct PipelineResult { run_id, status, completed, skipped, failed, checkpoint }
   
   // Cache — two-tier
   pub struct Cache { memory: moka::future::Cache, disk: Option<DiskCache>, default_policy }
